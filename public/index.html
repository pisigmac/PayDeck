import (
    "context"
    "os"
    "github.com/pisigmac/PayDeck/sdks/go/paydeck"
)

func main() {
    baseURL := os.Getenv("PAYDECK_BASE_URL")
    if baseURL == "" {
        baseURL = "https://api.paydeck.dev"
    }
    client := paydeck.NewClient(paydeck.ClientOptions{
        BaseURL: baseURL,
        APIKey:  "pd_live_12345...",
        Retries: 3,
    })

    order, err := client.CreateOrder(context.Background(), paydeck.CreateOrderInput{
        Plan:           "pro",
        IdempotencyKey: "sess_100",
    })
}
